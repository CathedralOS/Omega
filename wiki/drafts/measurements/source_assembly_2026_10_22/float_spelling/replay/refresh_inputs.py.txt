from pathlib import Path
import json,subprocess,hashlib,shutil
p=Path('build/float-spelling');runtime=Path((p/'runtime-directory.txt').read_text().strip())
original=json.loads((p/'parser-input-hashes.json').read_text());inputs=[{'path':row['path'],'sha256':hashlib.sha256(Path(row['path']).read_bytes()).hexdigest()} for row in original]
changed=[row['path'] for row,new in zip(original,inputs) if row['sha256']!=new['sha256']]
binary_hashes={side:hashlib.sha256((p/f'probe-{side}').read_bytes()).hexdigest() for side in ['baseline','candidate']}
for existing in (p/'integrations').glob('*/parser-comparison.json'):
 if existing.parent==runtime:continue
 prior=json.loads(existing.read_text());prior_inputs=json.loads((existing.parent/'parser-input-hashes.json').read_text())
 if prior_inputs!=inputs or prior.get('binary_hashes')!=binary_hashes:continue
 prior['reused_from']=str(existing)
 prior['reuse_reason']='All 463 input byte hashes and both frozen probe hashes are unchanged; no new parser observation is claimed.'
 (runtime/'parser-comparison.json').write_text(json.dumps(prior,indent=2)+'\n')
 shutil.copy2(existing.parent/'parser-input-hashes.json',runtime/'parser-input-hashes.json')
 audit=json.loads((p/'parser-evidence-applicability.json').read_text());audit['changed_measured_inputs']=changed;audit['updated_input_comparison']=str(runtime/'parser-comparison.json');(p/'parser-evidence-applicability.json').write_text(json.dumps(audit,indent=2)+'\n')
 print('Reused updated parser comparison:',existing,flush=True)
 raise SystemExit(0)
results={}
for side in ['baseline','candidate']:
 with (runtime/f'parser-{side}.json').open('w') as output:subprocess.run([str(p/f'probe-{side}'),str(p/'syntax-inputs.txt')],stdout=output,check=True)
 results[side]=json.loads((runtime/f'parser-{side}.json').read_text())['inputs']
assert len(results['baseline'])==len(results['candidate'])==len(inputs)==463
summary={}
for side,rows in results.items():
 summary[side]={key:sum(row.get('metrics',{}).get(key,0) for row in rows) for key in ['allocation_requests','requested_bytes','live_requested_bytes','peak_live_requested_bytes']}
 summary[side]['retained_source_references']=sum(row.get('retained_syntax_source_references',0) for row in rows)
 summary[side]['successful_forests']=sum('syntax' in row.get('observation',{}) for row in rows)
 summary[side]['inputs']=len(rows)
summary['differing_observations']=[a['path'] for a,b in zip(results['baseline'],results['candidate']) if a['path']!=b['path'] or a.get('observation')!=b.get('observation')]
assert not summary['differing_observations'],summary['differing_observations']
summary['input_revision']=subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()
summary['original_input_paths_changed']=changed
summary['binary_revisions']={'baseline':(p/'base.txt').read_text().strip(),'candidate':'a38e680dfe48e488ce73852c8452d8185dad7e1e'}
summary['binary_hashes']={side:hashlib.sha256((p/f'probe-{side}').read_bytes()).hexdigest() for side in ['baseline','candidate']}
summary['applicability']='Original frozen parser probes replay the updated selected input corpus. Their complete parser dependency closure has no incoming changes; producer revisions remain historical. No timing was collected during this allocation/output replay.'
(runtime/'parser-comparison.json').write_text(json.dumps(summary,indent=2)+'\n');(runtime/'parser-input-hashes.json').write_text(json.dumps(inputs,indent=2)+'\n')
audit=json.loads((p/'parser-evidence-applicability.json').read_text());audit['changed_measured_inputs']=changed;audit['updated_input_comparison']=str(runtime/'parser-comparison.json');(p/'parser-evidence-applicability.json').write_text(json.dumps(audit,indent=2)+'\n')
print('Updated input comparison:',json.dumps(summary),flush=True)
