use syntax_trees::SyntaxTextContext;
fn normalize_debug(syntax: &syntax_trees::SyntaxTrees) -> String {
    let rendered = format!("{syntax:?}");
    let names = regex::Regex::new(r"Identifier \{ source: Handle \{ arena_index: 1, generation: (\d+) \}, source_span: (SourceSpan \{ source_id: SourceId\(\d+\), span: Span \{ start: (\d+), end: (\d+) \} \}) \}").unwrap();
    let rendered = names.replace_all(&rendered, |captures: &regex::Captures<'_>| {
        let source = syntax.source_text(arena::Handle::from_parts(1, captures[1].parse().unwrap()));
        let start: usize = captures[3].parse().unwrap();
        let end: usize = captures[4].parse().unwrap();
        format!("Identifier {{ text: {:?}, source_span: {} }}", &source[start..end], &captures[2])
    });
    let floats = regex::Regex::new(r"FloatSpelling \{ source: Handle \{ arena_index: 1, generation: (\d+) \}, source_span: (SourceSpan \{ source_id: SourceId\(\d+\), span: Span \{ start: (\d+), end: (\d+) \} \}) \}").unwrap();
    let rendered = floats.replace_all(&rendered, |captures: &regex::Captures<'_>| {
        let source = syntax.source_text(arena::Handle::from_parts(1, captures[1].parse().unwrap()));
        let start: usize = captures[3].parse().unwrap();
        let end: usize = captures[4].parse().unwrap();
        format!("SourceText {{ text: {:?}, source_span: {} }}", &source[start..end], &captures[2])
    });
    let rendered = rendered.replace("FloatSpelling { text:", "SourceText { text:");
    assert!(!rendered.contains("FloatSpelling { source:"), "unresolved float spelling");
    let owner = regex::Regex::new(r"SyntaxTreeTables \{ text: SyntaxTextOwner \{ source_count: \d+ \}, ").unwrap();
    assert_eq!(owner.find_iter(&rendered).count(), 1, "one syntax owner field");
    let normalized = owner.replace_all(&rendered, "SyntaxTreeTables { ").into_owned();
    assert!(!normalized.contains("Identifier { source:"), "unresolved name in semantic debug comparison");
    regex::Regex::new(r"revision: \d+").unwrap().replace_all(&normalized, "revision: <mutation-sequence>").into_owned()
}
