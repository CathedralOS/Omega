from pathlib import Path
import subprocess,json,statistics,os,hashlib
p=Path('build/float-spelling');records=[]
for round_number in range(4):
 for side in (['baseline','candidate'] if round_number%2==0 else ['candidate','baseline']):
  load_before=os.getloadavg()
  result=subprocess.run([str(p/f'timing-{side}'),str(p/'syntax-inputs.txt')],capture_output=True,text=True,check=True)
  observation=json.loads(result.stdout);records.append(dict(round=round_number,producer=side,load_before=load_before,load_after=os.getloadavg(),**observation))
 print('Timing pair',round_number+1,'complete',flush=True)
summary={}
for side in ['baseline','candidate']:
 values=[v/1e6 for row in records if row['producer']==side for v in row['observations_ns']]
 summary[side]=dict(min_ms=min(values),median_ms=statistics.median(values),mean_ms=statistics.mean(values),max_ms=max(values),trials=len(values))
report=dict(scope='parse and destroy prelexed source files; no lexer, allocator counters, snapshots or I/O inside timed interval',base=(p/'base.txt').read_text().strip(),binary_hashes={side:hashlib.sha256((p/f'timing-{side}').read_bytes()).hexdigest() for side in ['baseline','candidate']},summary=summary,records=records)
(p/'timing.json').write_text(json.dumps(report,indent=2)+'\n');print(json.dumps(summary),flush=True)
