use std::{hint::black_box, sync::Arc, time::{Duration, Instant}};
fn main() {
    std::thread::Builder::new().stack_size(256*1024*1024).spawn(run).unwrap().join().unwrap();
}
fn run() {
    let paths = std::fs::read_to_string(std::env::args().nth(1).unwrap()).unwrap();
    let sources: Vec<Arc<str>> = paths.lines().map(|path| Arc::from(std::fs::read_to_string(path).unwrap())).collect();
    let streams: Vec<_> = sources.iter().filter_map(|source| source_files_to_tokens::Lexer::new(source).tokenize().ok().map(|tokens| tokens.retain_source(source.clone()))).collect();
    let mut observations = Vec::new();
    let mut minimum = u128::MAX;
    let started = Instant::now();
    let mut last_improvement = started;
    while observations.len() < 20 || last_improvement.elapsed() < Duration::from_secs(3) {
        let trial = Instant::now();
        for stream in &streams {
            let parsed = tokens_to_syntax_trees::parse_syntax_trees(black_box(stream));
            black_box(&parsed);
            drop(parsed);
        }
        let elapsed = trial.elapsed().as_nanos();
        if elapsed < minimum { minimum = elapsed; last_improvement = Instant::now(); }
        observations.push(elapsed);
        if started.elapsed() > Duration::from_secs(15) { break; }
    }
    println!("{}", serde_json::json!({"inputs":streams.len(),"expression_node_size":size_of::<syntax_trees::expression::ExpressionNode>(),"minimum_ns":minimum,"quiet_reached":last_improvement.elapsed() >= Duration::from_secs(3),"observations_ns":observations}));
}
