import os,pathlib,subprocess,shutil
p=pathlib.Path('build/ingress-ownership');os.environ['OMEGA_LIBRARY_ROOT']=str(pathlib.Path('source/library').resolve());active=p/'assembly-active'
for name in ['normalized-baseline','candidate']:
 for mode,extra in [('unique',[]),('shared',['--shared']),('all',[])]:
  binary=p/f'harness-{name}{"-all" if mode=="all" else ""}'
  if active.exists():active.unlink()
  active.symlink_to(binary.resolve())
  for extension in ['txt','stderr']:
   original=p/f'{name}-{mode}.{extension}'
   if original.exists():shutil.copy2(original,p/f'original-argv-{name}-{mode}.{extension}')
  with (p/f'{name}-{mode}.txt').open('w') as out,(p/f'{name}-{mode}.stderr').open('w') as err:
   subprocess.run([str(active),str(p/'inputs.txt'),*extra],stdout=out,stderr=err,check=True)
 print(name,'stable invocation complete',flush=True)
active.unlink()
exec((p/'summarize_assembly.py').read_text())
