machine capacity<Storage, const Capacity: u64>() -> u64
where Storage == [u8; Capacity]
{ Capacity }

machine build(builder: &mut Build) {
    Build::application(builder, "extent_probe");
    let observed: u64 = capacity<[u8; 8], 8>();
    transition observed == 8 { true -> good(builder) false -> bad(builder) }
    state good(builder: &mut Build) {
        Build::bind(builder, macos_arm64::ProgramEntry, main);
        Build::bind(builder, linux_arm64::ProgramEntry, main);
        Build::bind(builder, linux_x86_64::ProgramEntry, main);
        Build::bind(builder, windows_x86_64::ProgramEntry, main);
    }
    state bad(builder: &mut Build) {
        _ = BuildProduct::entry(&builder.product, "incorrect_extent", "macos_arm64::ProgramEntry");
    }
}
