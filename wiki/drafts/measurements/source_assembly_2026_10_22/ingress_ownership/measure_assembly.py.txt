import subprocess, pathlib, os, json
p=pathlib.Path('build/ingress-ownership');os.environ['OMEGA_LIBRARY_ROOT']=str(pathlib.Path('source/library').resolve())
for name,manifest in [('normalized-baseline','baseline-harness'),('candidate','harness')]:
 with (p/f'{name}-harness-build.log').open('w') as out:
  subprocess.run(['mbx','build','--manifest-path',str(p/manifest/'Cargo.toml'),'--bins'],stdout=out,stderr=subprocess.STDOUT,check=True)
 for binary,suffix in [('assembly-measurement',''),('all_threads','-all')]:
  import shutil
  shutil.copy2(p/manifest/'target/debug'/binary,p/f'harness-{name}{suffix}')
 for mode,extra in [('unique',[]),('shared',['--shared']),('all',[])]:
  executable=p/f'harness-{name}{"-all" if mode=="all" else ""}'
  with (p/f'{name}-{mode}.txt').open('w') as out,(p/f'{name}-{mode}.stderr').open('w') as err:
   subprocess.run([str(executable),str(p/'inputs.txt'),*extra],stdout=out,stderr=err,check=True)
 print(name,'complete',flush=True)
results={}
for mode in ['unique','shared','all']:
 def read(name):return [line.split('\t') for line in (p/f'{name}-{mode}.txt').read_text().splitlines()]
 a,b=read('normalized-baseline'),read('candidate');assert len(a)==len(b)==289
 results[mode]={'baseline_calls':sum(int(r[0]) for r in a),'candidate_calls':sum(int(r[0]) for r in b),'baseline_bytes':sum(int(r[1]) for r in a),'candidate_bytes':sum(int(r[1]) for r in b),'different':[(x[5],x[3],y[3]) for x,y in zip(a,b) if x[3:]!=y[3:]]}
print(json.dumps(results,indent=2),flush=True);(p/'assembly-comparison.json').write_text(json.dumps(results,indent=2)+'\n')
