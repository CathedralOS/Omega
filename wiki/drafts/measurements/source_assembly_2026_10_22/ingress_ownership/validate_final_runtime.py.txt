import json,subprocess,os
from pathlib import Path
p=Path('build/ingress-ownership')
os.environ['OMEGA_LIBRARY_ROOT']=str(Path('source/library').resolve())
base=(p/'comparison-base.txt').read_text().strip();selection=(p/'corpus-filter.txt').read_text().strip();results=[]
for leg,extra in [('check',[]),('both',['--both'])]:
 command=['python3','tools/corpus_gate.py','--runner',str((p/'corpus-runner-final-candidate').resolve()),'--compare',base,'--filter',selection,'--json',str(p/f'corpus-{leg}.json'),*extra]
 with (p/f'corpus-{leg}.log').open('w') as out: result=subprocess.run(command,stdout=out,stderr=subprocess.STDOUT)
 report=json.loads((p/f'corpus-{leg}.json').read_text())
 assert report['schema']=='omega-corpus-compare-v1' and result.returncode==0,report
 assert not report['losses'] and not report['gains'],report
 results.append(dict(command=command,exit=result.returncode,comparison=report))
 # The compare verdict omits raw engine observations. Capture a read-only
 # verdict against the published baseline golden, without stamping candidate rows.
 record_command=['python3','tools/corpus_gate.py','--runner',str((p/'corpus-runner-final-candidate').resolve()),'--filter',selection,'--golden',str(Path('build/corpus_baselines')/(base+('-both' if extra else '')+'.txt')),'--json',str(p/f'candidate-record-{leg}.json'),*extra]
 with (p/f'candidate-record-{leg}.log').open('w') as out:recorded=subprocess.run(record_command,stdout=out,stderr=subprocess.STDOUT)
 record=json.loads((p/f'candidate-record-{leg}.json').read_text())
 assert record['infrastructure_error'] is None and recorded.returncode in (0,1,3),record
 baseline=json.loads((p/f'baseline-record-{leg}.json').read_text())
 def observed(row):return {k:row[k] for k in ['fixture','status','facts','diagnostics']}
 assert [observed(row) for row in record['observations']]==[observed(row) for row in baseline['observations']],leg
 results.append(dict(command=record_command,exit=recorded.returncode,exact_baseline_observations_match=True))
 print(leg,result.returncode,'no losses or gains; complete recorded observations match baseline',flush=True)
cli_rows=[]
for fixture in ['build/captured_source_generated_constant','build/generated_source_selects_authored','build/dependency_declaration_queries']:
 observations=[]
 for producer in ['baseline','final-candidate']:
  command=[str(p/f'omega-{producer}'),'--check',f'tests/omega/pass/{fixture}/main.omg']
  result=subprocess.run(command,text=True,capture_output=True)
  row=dict(fixture=fixture,producer=producer,command=command,exit=result.returncode,stdout=result.stdout,stderr=result.stderr);cli_rows.append(row);observations.append((result.returncode,result.stdout,result.stderr))
 assert observations[0]==observations[1],(fixture,observations)
(p/'cli-build-validation.json').write_text(json.dumps(cli_rows,indent=2)+'\n')
print('CLI Build comparisons',[(r['fixture'],r['producer'],r['exit']) for r in cli_rows],flush=True)
command=[str(p/'omega-final-candidate'),'run','--both','samples/cli/facts/length/local/main.omg']
with (p/'length-both.stdout').open('w') as out,(p/'length-both.stderr').open('w') as err:result=subprocess.run(command,stdout=out,stderr=err)
assert result.returncode==0
results.append(dict(command=command,exit=result.returncode))
invalid=[]
for producer in ['baseline','final-candidate']:
 command=[str(p/f'omega-{producer}'),'--check','samples/cli/facts/length/local/invalid/main.omg']
 result=subprocess.run(command,text=True,capture_output=True)
 invalid.append(dict(producer=producer,command=command,exit=result.returncode,stdout=result.stdout,stderr=result.stderr))
assert invalid[0]['exit']!=0 and all(invalid[0][key]==invalid[1][key] for key in ['exit','stdout','stderr']),invalid
(p/'length-invalid.json').write_text(json.dumps(invalid,indent=2)+'\n')
results.append(dict(length_invalid_matches_baseline=True,exit=invalid[1]['exit']))
(p/'runtime-validation.json').write_text(json.dumps(results,indent=2)+'\n')
print('length both 0; invalid control matches baseline rejection',flush=True)
