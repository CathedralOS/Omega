use std::{alloc::{GlobalAlloc,Layout},cell::Cell,hint::black_box,path::PathBuf,sync::Arc,time::{Duration,Instant}};
use source::{LoadedSource,SourceId};
use syntax_trees::SyntaxTrees;
thread_local! {static COUNTS:Cell<(bool,u64,u64)> = const { Cell::new((false,0,0)) };}
struct Allocator;
fn count(bytes:usize) {COUNTS.with(|state| {let (enabled,calls,total)=state.get();if enabled {state.set((true,calls+1,total+bytes as u64));}});}
unsafe impl GlobalAlloc for Allocator {
 unsafe fn alloc(&self,layout:Layout)->*mut u8 {count(layout.size());unsafe {mimalloc::MiMalloc.alloc(layout)}}
 unsafe fn alloc_zeroed(&self,layout:Layout)->*mut u8 {count(layout.size());unsafe {mimalloc::MiMalloc.alloc_zeroed(layout)}}
 unsafe fn dealloc(&self,pointer:*mut u8,layout:Layout) {unsafe {mimalloc::MiMalloc.dealloc(pointer,layout)}}
 unsafe fn realloc(&self,pointer:*mut u8,layout:Layout,size:usize)->*mut u8 {count(size);unsafe {mimalloc::MiMalloc.realloc(pointer,layout,size)}}
}
#[global_allocator] static ALLOCATOR:Allocator=Allocator;
fn faults()->i64 {unsafe {let mut usage=std::mem::zeroed::<libc::rusage>();assert_eq!(libc::getrusage(libc::RUSAGE_SELF,&mut usage),0);usage.ru_minflt as i64}}
#[inline(never)] fn construct()->SyntaxTrees {SyntaxTrees::new(black_box(SourceId(1)))}
fn repeat<T>(label:&str,bytes:usize,quiet:f64,mut operation:impl FnMut()->T) {
 let mut minimum=Duration::MAX;let mut maximum=Duration::ZERO;let mut sum=Duration::ZERO;let mut trials=0;
 let beginning=Instant::now();let mut improved=beginning;let initial_faults=faults();
 while improved.elapsed().as_secs_f64()<quiet && beginning.elapsed().as_secs()<45 {
  let started=Instant::now();let output=operation();black_box(&output);let elapsed=started.elapsed();
  sum+=elapsed;maximum=maximum.max(elapsed);trials+=1;
  if elapsed<minimum {minimum=elapsed;improved=Instant::now();}
  drop(output);
 }
 println!("{label}: trials={trials} bytes={bytes} min_us={:.3} avg_us={:.3} max_us={:.3} minor_faults={} quiet_reached={}",minimum.as_secs_f64()*1e6,sum.as_secs_f64()*1e6/trials as f64,maximum.as_secs_f64()*1e6,faults()-initial_faults,improved.elapsed().as_secs_f64()>=quiet);
}
fn main(){
 let arguments:Vec<_>=std::env::args().collect();let quiet:f64=arguments[1].parse().unwrap();
 let files=std::fs::read_to_string(&arguments[2]).unwrap();
 let loaded=files.lines().enumerate().map(|(index,path)|LoadedSource{source_id:SourceId(index+1),path:PathBuf::from(path),source:Arc::from(std::fs::read_to_string(path).unwrap()),origin:None,scope:source::DependencyScope::Product}).collect();
 let lexed=source_files_to_tokens::lex_sources(loaded).unwrap();let sources=lexed.sources.span_or_empty(lexed.batch);
 let bytes:usize=sources.iter().map(|source|source.source.len()).sum();let tokens:usize=sources.iter().map(|source|source.tokens.len()).sum();
 println!("files={} bytes={bytes} tokens={tokens}",sources.len());
 COUNTS.with(|state|state.set((true,0,0)));let empty=construct();let empty_counts=COUNTS.with(|state| {let result=state.get();state.set((false,0,0));result});drop(empty);println!("empty counts: {empty_counts:?}");
 let mut digest=0xcbf29ce484222325u64;
 for source in sources {
  COUNTS.with(|state|state.set((true,0,0)));
  let syntax=tokens_to_syntax_trees::parse_syntax_trees_with_id(source.source_id,black_box(&source.tokens)).unwrap();
  let counts=COUNTS.with(|state| {let result=state.get();state.set((false,0,0));result});
  println!("counts {}: calls={} bytes={} roots={}",source.path.display(),counts.1,counts.2,syntax.roots.items.len());
  for byte in format!("{syntax:?}").bytes() {digest=(digest^u64::from(byte)).wrapping_mul(0x100000001b3);}
 }
 println!("syntax_digest={digest:016x}");
 repeat("empty syntax",0,quiet,construct);
 repeat("parse retained tokens",bytes,quiet,||sources.iter().map(|source|tokens_to_syntax_trees::parse_syntax_trees_with_id(source.source_id,black_box(&source.tokens)).unwrap()).collect::<Vec<_>>());
}
