use std::{alloc::{GlobalAlloc,Layout},cell::Cell,hint::black_box,path::PathBuf,sync::Arc,};
use source::{LoadedSource,SourceId};
thread_local! {static COUNTS:Cell<(bool,u64,u64)> = const { Cell::new((false,0,0)) };}
struct Allocator;
fn count(bytes:usize) {COUNTS.with(|state| {let (enabled,calls,total)=state.get();if enabled {state.set((true,calls+1,total+bytes as u64));}});}
unsafe impl GlobalAlloc for Allocator {
 unsafe fn alloc(&self,layout:Layout)->*mut u8 {count(layout.size());unsafe {mimalloc::MiMalloc.alloc(layout)}}
 unsafe fn alloc_zeroed(&self,layout:Layout)->*mut u8 {count(layout.size());unsafe {mimalloc::MiMalloc.alloc_zeroed(layout)}}
 unsafe fn dealloc(&self,pointer:*mut u8,layout:Layout) {unsafe {mimalloc::MiMalloc.dealloc(pointer,layout)}}
 unsafe fn realloc(&self,pointer:*mut u8,layout:Layout,size:usize)->*mut u8 {count(size);unsafe {mimalloc::MiMalloc.realloc(pointer,layout,size)}}
}
#[global_allocator] static ALLOCATOR:Allocator=Allocator;
fn main(){std::thread::Builder::new().stack_size(256*1024*1024).spawn(record).unwrap().join().unwrap();}
fn record(){
 let files=std::fs::read_to_string(std::env::args().nth(1).unwrap()).unwrap();
 for path in files.lines(){
  let loaded=vec![LoadedSource{source_id:SourceId(1),path:PathBuf::from(path),source:Arc::from(std::fs::read_to_string(path).unwrap()),origin:None,scope:source::DependencyScope::Product}];
  let Ok(lexed)=source_files_to_tokens::lex_sources(loaded.into_iter().collect()) else {continue};let source=&lexed.sources.span_or_empty(lexed.batch)[0];
  COUNTS.with(|state|state.set((true,0,0)));
  let result=tokens_to_syntax_trees::parse_syntax_trees_with_id(source.source_id,black_box(&source.tokens));black_box(&result);
  let counts=COUNTS.with(|state|{let result=state.get();state.set((false,0,0));result});
  println!("{}\t{}\t{}\t{}",counts.1,counts.2,if result.is_ok(){"parsed"}else{"rejected"},path);
 }
}
