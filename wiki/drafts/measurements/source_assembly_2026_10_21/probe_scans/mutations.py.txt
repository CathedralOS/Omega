from pathlib import Path
import subprocess,json
out=Path('build/probe-scans');folder=out/'controls';folder.mkdir(exist_ok=True)
# Both admissible grouped uses and malformed boundary/closer arrangements.
arguments=['Value','1','-1','true','1 >> 2','(value == 1)','match value { 0 -> 1, _ -> 2 }','(Record { value: 3 }).value + 1','[u8; 3]',"Wrapper<'region, Value>"]
separators=[' = next',': next',' -> next','; next','',' /* : = -> */ ']
closers=['>','>>','> >']
files=[]
for argument in arguments:
 for separator in separators:
  for closer in closers:
   p=folder/f'{len(files):03}.omg';p.write_text(f'machine caller() {{ callee<{argument}{separator}{closer}(); later<Value>(); }}\n');files.append(str(p.resolve()))
listing=out/'control-inputs.txt';listing.write_text('\n'.join(files)+'\n')
for version in ['baseline','candidate']:
 with (out/f'{version}-controls.tsv').open('w') as log:subprocess.run([str(out/f'parse-record-{version}'),str(listing)],stdout=log,check=True)
a=(out/'baseline-controls.tsv').read_bytes();b=(out/'candidate-controls.tsv').read_bytes();assert a==b
print('controls',len(files),'parsed',a.count(b'\tparsed\t'),'rejected',a.count(b'\trejected\t'))
