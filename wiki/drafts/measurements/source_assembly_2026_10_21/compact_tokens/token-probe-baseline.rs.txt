use std::alloc::{GlobalAlloc,Layout,System};
use std::sync::{Arc,atomic::{AtomicBool,AtomicUsize,Ordering}};
use tokens::{Token,TokenText};
use serde_json::json;
struct CountingAllocator;
static ENABLED: AtomicBool = AtomicBool::new(false);
static CALLS: AtomicUsize = AtomicUsize::new(0);
static REQUESTED: AtomicUsize = AtomicUsize::new(0);
static FREED: AtomicUsize = AtomicUsize::new(0);
static LIVE: AtomicUsize = AtomicUsize::new(0);
static PEAK: AtomicUsize = AtomicUsize::new(0);
fn add(bytes: usize) {
    CALLS.fetch_add(1, Ordering::Relaxed);
    REQUESTED.fetch_add(bytes, Ordering::Relaxed);
    let current = LIVE.fetch_add(bytes, Ordering::Relaxed) + bytes;
    PEAK.fetch_max(current, Ordering::Relaxed);
}
fn remove(bytes: usize) {
    FREED.fetch_add(bytes, Ordering::Relaxed);
    LIVE.fetch_sub(bytes, Ordering::Relaxed);
}
unsafe impl GlobalAlloc for CountingAllocator {
    unsafe fn alloc(&self, layout: Layout) -> *mut u8 {
        let pointer = unsafe { System.alloc(layout) };
        if !pointer.is_null() && ENABLED.load(Ordering::Relaxed) { add(layout.size()); }
        pointer
    }
    unsafe fn alloc_zeroed(&self, layout: Layout) -> *mut u8 {
        let pointer = unsafe { System.alloc_zeroed(layout) };
        if !pointer.is_null() && ENABLED.load(Ordering::Relaxed) { add(layout.size()); }
        pointer
    }
    unsafe fn dealloc(&self, pointer: *mut u8, layout: Layout) {
        if ENABLED.load(Ordering::Relaxed) { remove(layout.size()); }
        unsafe { System.dealloc(pointer, layout) };
    }
    unsafe fn realloc(&self, pointer: *mut u8, layout: Layout, size: usize) -> *mut u8 {
        let result = unsafe { System.realloc(pointer, layout, size) };
        if !result.is_null() && ENABLED.load(Ordering::Relaxed) { remove(layout.size()); add(size); }
        result
    }
}
#[global_allocator]
static ALLOCATOR: CountingAllocator = CountingAllocator;
fn begin() {
    for counter in [&CALLS, &REQUESTED, &FREED, &LIVE, &PEAK] { counter.store(0, Ordering::Relaxed); }
    ENABLED.store(true, Ordering::Relaxed);
}
fn finish() -> serde_json::Value {
    ENABLED.store(false, Ordering::Relaxed);
    json!({"allocation_requests":CALLS.load(Ordering::Relaxed),"requested_bytes":REQUESTED.load(Ordering::Relaxed),"live_requested_bytes":LIVE.load(Ordering::Relaxed),"peak_live_requested_bytes":PEAK.load(Ordering::Relaxed)})
}
fn retain(tokens: tokens::TokenStream<'_>, source: &Arc<str>) -> Vec<Token<'static>> {
    tokens.into_tokens().into_iter().map(|token| {
        let lexeme=match token.lexeme {
            TokenText::Source(_) => TokenText::shared(source.clone(), token.span),
            TokenText::Shared {source,span} => TokenText::Shared {source,span},
            TokenText::Owned(value) => TokenText::Owned(value),
            TokenText::OwnedBytes(value) => TokenText::OwnedBytes(value),
        };
        Token {kind:token.kind,lexeme,span:token.span}
    }).collect()
}

fn main() {
 let mut output=Vec::new();
 for path in std::fs::read_to_string(std::env::args().nth(1).unwrap()).unwrap().lines() {
  let source:Arc<str>=Arc::from(std::fs::read_to_string(path).unwrap());
  begin();
  let result=source_files_to_tokens::Lexer::new(&source).tokenize();
  let result=result.map(|tokens| retain(tokens,&source));
  std::hint::black_box(&result);
  let metrics=finish();
  let references=Arc::strong_count(&source)-1;
  let observation=match result {
   Ok(tokens)=>{let logical=tokens.iter().map(|token|format!("{:?}:{:?}:{:?}",token.kind,token.span,token.lexeme.as_bytes())).collect::<Vec<_>>();json!({"tokens":tokens.len(),"logical":logical})},
   Err(error)=>json!({"error":format!("{error:?}")}),
  };
  output.push(json!({"path":path,"metrics":metrics,"source_references":references,"observation":observation}));
 }
 println!("{}",serde_json::to_string(&json!({"token_size":size_of::<Token<'static>>(),"inputs":output})).unwrap());
}
