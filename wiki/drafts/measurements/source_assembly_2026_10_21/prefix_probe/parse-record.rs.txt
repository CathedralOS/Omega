use std::{path::PathBuf,sync::Arc};
use source::{LoadedSource,SourceId};
use sha2::{Digest,Sha256};
#[global_allocator] static ALLOCATOR:mimalloc::MiMalloc=mimalloc::MiMalloc;
fn main(){ std::thread::Builder::new().stack_size(256*1024*1024).spawn(record).unwrap().join().unwrap(); }
fn record(){
 let list=std::fs::read_to_string(std::env::args().nth(1).unwrap()).unwrap();
 for path in list.lines(){
  let source=std::fs::read_to_string(path).unwrap();
  let loaded=[LoadedSource{source_id:SourceId(1),path:PathBuf::from(path),source:Arc::from(source),origin:None,scope:source::DependencyScope::Product}].into_iter().collect();
  let result=match source_files_to_tokens::lex_sources(loaded){
   Ok(lexed)=>{let source=&lexed.sources.span_or_empty(lexed.batch)[0];format!("{:?}",tokens_to_syntax_trees::parse_syntax_trees_with_id(source.source_id,&source.tokens))},
   Err(error)=>format!("lexical {error:?}"),
  };
  println!("{:x}\t{}\t{}",Sha256::digest(result.as_bytes()),if result.starts_with("Ok("){"parsed"}else{"rejected"},path);
 }
}
