use std::{alloc::{GlobalAlloc, Layout}, cell::Cell, hint::black_box, path::PathBuf, sync::Arc, time::{Duration, Instant}};
use source::{DependencyScope, SourceMap, SourceOrigin, SourceResolutionStratum};
thread_local! { static COUNTS: Cell<(bool,u64,u64)> = const { Cell::new((false,0,0)) }; }
struct Allocator;
fn count(bytes:usize) { COUNTS.with(|state| { let (enabled,calls,total)=state.get(); if enabled {state.set((true,calls+1,total+bytes as u64));} }); }
unsafe impl GlobalAlloc for Allocator {
 unsafe fn alloc(&self,layout:Layout)->*mut u8 {count(layout.size());unsafe {mimalloc::MiMalloc.alloc(layout)}}
 unsafe fn alloc_zeroed(&self,layout:Layout)->*mut u8 {count(layout.size());unsafe {mimalloc::MiMalloc.alloc_zeroed(layout)}}
 unsafe fn dealloc(&self,pointer:*mut u8,layout:Layout) {unsafe {mimalloc::MiMalloc.dealloc(pointer,layout)}}
 unsafe fn realloc(&self,pointer:*mut u8,layout:Layout,size:usize)->*mut u8 {count(size);unsafe {mimalloc::MiMalloc.realloc(pointer,layout,size)}}
}
#[global_allocator] static ALLOCATOR:Allocator=Allocator;
#[inline(never)]
fn copied(inputs:&[(PathBuf,Arc<str>)])->SourceMap {
 let mut map=SourceMap::default();
 for (path,text) in inputs {
  map.add_checked_instance(path.clone(),text.to_string(),path.parent().unwrap().to_path_buf(),None,SourceOrigin::User,SourceResolutionStratum::Base,DependencyScope::Product);
 }
 map
}
#[inline(never)]
fn retained(inputs:&[(PathBuf,Arc<str>)])->SourceMap {
 let mut map=SourceMap::default();
 for (path,text) in inputs {
  map.add_checked_instance(path.clone(),Arc::clone(text),path.parent().unwrap().to_path_buf(),None,SourceOrigin::User,SourceResolutionStratum::Base,DependencyScope::Product);
 }
 map
}
fn repeat(label:&str,inputs:&[(PathBuf,Arc<str>)],operation:fn(&[(PathBuf,Arc<str>)])->SourceMap) {
 COUNTS.with(|state|state.set((true,0,0)));
 let map=operation(black_box(inputs)); black_box(&map);
 let (_,calls,bytes)=COUNTS.with(|state| {let value=state.get();state.set((false,0,0));value});drop(map);
 let mut minimum=Duration::MAX;let mut maximum=Duration::ZERO;let mut sum=Duration::ZERO;let mut trials=0u64;
 let beginning=Instant::now();let mut improved=beginning;
 while improved.elapsed()<Duration::from_secs(2) && beginning.elapsed()<Duration::from_secs(20) {
  let started=Instant::now();let map=operation(black_box(inputs));black_box(&map);let elapsed=started.elapsed();
  sum+=elapsed;maximum=maximum.max(elapsed);trials+=1;
  if elapsed<minimum {minimum=elapsed;improved=Instant::now();}
  drop(map);
 }
 println!("{label}: calls={calls} bytes={bytes} trials={trials} min_ns={} mean_ns={} max_ns={} quiet={}",minimum.as_nanos(),sum.as_nanos()/u128::from(trials),maximum.as_nanos(),improved.elapsed()>=Duration::from_secs(2));
}
fn main() {
 let paths=std::fs::read_to_string(std::env::args().nth(1).unwrap()).unwrap();
 let inputs:Vec<_>=paths.lines().map(|path|(PathBuf::from(path),Arc::<str>::from(std::fs::read_to_string(path).unwrap()))).collect();
 assert_eq!(copied(&inputs),retained(&inputs));
 println!("files={} source_bytes={}",inputs.len(),inputs.iter().map(|(_,text)|text.len()).sum::<usize>());
 for (label,operation) in [("copied",copied as fn(&[(PathBuf,Arc<str>)])->SourceMap),("retained",retained),("retained",retained),("copied",copied)] {repeat(label,&inputs,operation);}
}
