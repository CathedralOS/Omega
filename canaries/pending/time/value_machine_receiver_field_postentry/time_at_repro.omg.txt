// std::time — the Duration pure-value core (TASKS_TIME.md rung 3, D1-D5/D13).
//
// `Duration` is a normalized span: whole seconds plus subsecond nanoseconds,
// where the field RANGE carries the normalization invariant the compiler
// write-enforces at every construction site — ZII zero IS the zero duration.
// Everything here is a pure `[copy, zero_init]` value with no authority; clock
// READS live on the `Time` capability (a later rung), never on these types.
//
// Rust parity, differing where Omega is better (TASKS_TIME.md north star):
// - No panicking `+`/`-`: the checked and saturating forms ARE the API.
//   `case Overflow` is DurationResult's ZII-zero first case, so an
//   uninitialized result reads as the failure case, never a fake success.
// - `checked_subtract` reports underflow through the same `Overflow` case
//   (one result vocabulary; the direction is unambiguous from the operation).
// - Construction is by struct literal today (`Duration { seconds: 2,
//   subsecond_nanoseconds: 0 }`); `from_*` unit constructors and the
//   `MAX`/saturating_add ceiling arrive with the next D14 gate position
//   (u64-magnitude literals in struct-literal fields).
//
// Authoring shape (the de-Trapping house recipes):
// - Overflow/underflow DETECTION is explicit guards; the arithmetic itself
//   runs in Wrapping-domained lets (obligation-free), whose values are only
//   USED on the guard arms that prove them meaningful — the wrapped-compare
//   idiom (`sum < a` iff the u64 add wrapped) needs no u64::MAX literal.
// - The ranged `subsecond_nanoseconds` store happens in `*_emit` states whose
//   SINGLE inbound edge is the `<= 999999999` guard: single-predecessor edge
//   guards flow into the target state, so the store proves. The `false` arms
//   are unreachable by construction but total (never a silent face).

data Ordering [copy, zero_init] {
    case Less;
    case Equal;
    case Greater;
}

data Duration [copy, zero_init] {
    seconds: u64;
    subsecond_nanoseconds: u32 [0..=999_999_999];
}

data DurationResult [copy, zero_init] {
    case Overflow;
    case Ok(duration: Duration);
}

const Duration::ZERO: Duration = Duration { seconds: 0, subsecond_nanoseconds: 0 };

// Whole seconds, truncating the subsecond part (Rust `as_secs`).
machine Duration::as_seconds(&self) -> u64 {
    self.seconds
}

machine Duration::is_zero(&self) -> bool {
    self.seconds == 0 && self.subsecond_nanoseconds == 0
}

// self + other; `Overflow` when the true sum exceeds the representable range.
machine Duration::checked_add(&self, other: Duration) -> DurationResult {
    // EXACT-provable straight from the field ranges:
    // [0..=999_999_999] + [0..=999_999_999] = [0..=1_999_999_998] fits u32.
    let nanoseconds_total: u32 = self.subsecond_nanoseconds + other.subsecond_nanoseconds;
    let seconds_sum: u64 in Wrapping =
        (self.seconds as u64 in Wrapping) + (other.seconds as u64 in Wrapping);
    let seconds_wrapped: bool = seconds_sum < self.seconds;
    transition nanoseconds_total >= 1000000000 {
        true -> add_carried(nanoseconds_total, seconds_sum, seconds_wrapped)
        false -> add_flat(nanoseconds_total, seconds_sum, seconds_wrapped)
    }

    state add_carried(&self, nanoseconds_total: u32, seconds_sum: u64, seconds_wrapped: bool) -> DurationResult {
        let seconds_carried: u64 in Wrapping = (seconds_sum as u64 in Wrapping) + 1;
        let carry_wrapped: bool = seconds_carried < seconds_sum;
        let nanoseconds_reduced: u32 in Wrapping = (nanoseconds_total as u32 in Wrapping) - 1000000000;
        transition seconds_wrapped {
            true -> add_overflow()
            false -> add_carry_gate(nanoseconds_reduced, seconds_carried, carry_wrapped)
        }
    }
    state add_carry_gate(&self, nanoseconds_reduced: u32, seconds_carried: u64, carry_wrapped: bool) -> DurationResult {
        transition carry_wrapped {
            true -> add_overflow()
            false -> add_gate(seconds_carried, nanoseconds_reduced)
        }
    }
    state add_flat(&self, nanoseconds_total: u32, seconds_sum: u64, seconds_wrapped: bool) -> DurationResult {
        transition seconds_wrapped {
            true -> add_overflow()
            false -> add_gate(seconds_sum, nanoseconds_total)
        }
    }
    state add_gate(&self, seconds_final: u64, nanoseconds_final: u32) -> DurationResult {
        transition nanoseconds_final <= 999999999 {
            true -> add_emit(seconds_final, nanoseconds_final)
            false -> add_overflow()
        }
    }
    state add_emit(&self, seconds_final: u64, nanoseconds_final: u32) -> DurationResult {
        // `% 1000000000` INLINE in the field value: construction checks are
        // flow-insensitive, and modulo bounds by the divisor, so the ranged
        // store proves right here; the gate already ensured the mod is an
        // identity at runtime.
        DurationResult::Ok {
            duration: Duration {
                seconds: seconds_final,
                subsecond_nanoseconds: nanoseconds_final % 1000000000,
            },
        }
    }
    state add_overflow(&self) -> DurationResult {
        DurationResult::Overflow
    }
}

// self - other; `Overflow` (underflow) when other exceeds self.
machine Duration::checked_subtract(&self, other: Duration) -> DurationResult {
    // Wrapping scratch: only the guard arms that prove no real underflow ever
    // USE these values.
    let seconds_difference: u64 in Wrapping =
        (self.seconds as u64 in Wrapping) - (other.seconds as u64 in Wrapping);
    let seconds_borrowed: u64 in Wrapping = (seconds_difference as u64 in Wrapping) - 1;
    let nanoseconds_direct: u32 in Wrapping = (self.subsecond_nanoseconds as u32 in Wrapping)
        - (other.subsecond_nanoseconds as u32 in Wrapping);
    let nanoseconds_borrowed: u32 in Wrapping = ((1000000000 as u32 in Wrapping)
        - (other.subsecond_nanoseconds as u32 in Wrapping))
        + (self.subsecond_nanoseconds as u32 in Wrapping);
    transition self.seconds > other.seconds {
        true -> subtract_bigger(seconds_difference, seconds_borrowed, nanoseconds_direct, nanoseconds_borrowed)
        false -> subtract_equal_or_less(seconds_difference, seconds_borrowed, nanoseconds_direct, nanoseconds_borrowed)
    }

    state subtract_bigger(&self, seconds_difference: u64, seconds_borrowed: u64, nanoseconds_direct: u32, nanoseconds_borrowed: u32) -> DurationResult {
        // Seconds strictly bigger: a nanosecond borrow cannot underflow.
        transition self.subsecond_nanoseconds >= other.subsecond_nanoseconds {
            true -> subtract_gate(seconds_difference, nanoseconds_direct)
            false -> subtract_gate_borrowed(seconds_borrowed, nanoseconds_borrowed)
        }
    }
    state subtract_equal_or_less(&self, seconds_difference: u64, seconds_borrowed: u64, nanoseconds_direct: u32, nanoseconds_borrowed: u32) -> DurationResult {
        transition self.seconds == other.seconds && self.subsecond_nanoseconds >= other.subsecond_nanoseconds {
            true -> subtract_gate_direct(seconds_difference, nanoseconds_direct)
            false -> subtract_underflow()
        }
    }
    state subtract_gate(&self, seconds_final: u64, nanoseconds_final: u32) -> DurationResult {
        transition nanoseconds_final <= 999999999 {
            true -> subtract_emit(seconds_final, nanoseconds_final)
            false -> subtract_underflow()
        }
    }
    state subtract_gate_borrowed(&self, seconds_final: u64, nanoseconds_final: u32) -> DurationResult {
        transition nanoseconds_final <= 999999999 {
            true -> subtract_emit(seconds_final, nanoseconds_final)
            false -> subtract_underflow()
        }
    }
    state subtract_gate_direct(&self, seconds_final: u64, nanoseconds_final: u32) -> DurationResult {
        transition nanoseconds_final <= 999999999 {
            true -> subtract_emit(seconds_final, nanoseconds_final)
            false -> subtract_underflow()
        }
    }
    state subtract_emit(&self, seconds_final: u64, nanoseconds_final: u32) -> DurationResult {
        // Same inline `%`-bounded store shape as add_emit (see there).
        DurationResult::Ok {
            duration: Duration {
                seconds: seconds_final,
                subsecond_nanoseconds: nanoseconds_final % 1000000000,
            },
        }
    }
    state subtract_underflow(&self) -> DurationResult {
        DurationResult::Overflow
    }
}

// self - other, clamping at ZERO instead of reporting underflow (total, the
// ZII-friendly default the wrapper types lean on).
machine Duration::saturating_subtract(&self, other: Duration) -> Duration {
    let difference: DurationResult = self.checked_subtract(other);
    transition difference {
        DurationResult::Ok { duration } -> saturate_value(duration)
        DurationResult::Overflow -> saturate_zero()
    }
    state saturate_value(&self, duration: Duration) -> Duration { duration }
    state saturate_zero(&self) -> Duration { Duration::ZERO }
}

machine Duration::compare(&self, other: Duration) -> Ordering {
    transition self.seconds > other.seconds {
        true -> compare_greater()
        false -> compare_seconds_rest(other)
    }
    state compare_seconds_rest(&self, other: Duration) -> Ordering {
        transition self.seconds < other.seconds {
            true -> compare_less()
            false -> compare_nanoseconds(other)
        }
    }
    state compare_nanoseconds(&self, other: Duration) -> Ordering {
        transition self.subsecond_nanoseconds > other.subsecond_nanoseconds {
            true -> compare_greater()
            false -> compare_nanoseconds_rest(other)
        }
    }
    state compare_nanoseconds_rest(&self, other: Duration) -> Ordering {
        transition self.subsecond_nanoseconds < other.subsecond_nanoseconds {
            true -> compare_less()
            false -> compare_equal()
        }
    }
    state compare_less(&self) -> Ordering { Ordering::Less }
    state compare_equal(&self) -> Ordering { Ordering::Equal }
    state compare_greater(&self) -> Ordering { Ordering::Greater }
}

machine Duration::is_less_than(&self, other: Duration) -> bool {
    let order: Ordering = self.compare(other);
    transition order {
        Ordering::Less -> less_yes()
        Ordering::Equal -> less_no()
        Ordering::Greater -> less_no()
    }
    state less_yes(&self) -> bool { true }
    state less_no(&self) -> bool { false }
}

machine Duration::is_greater_than(&self, other: Duration) -> bool {
    let order: Ordering = self.compare(other);
    transition order {
        Ordering::Greater -> greater_yes()
        Ordering::Less -> greater_no()
        Ordering::Equal -> greater_no()
    }
    state greater_yes(&self) -> bool { true }
    state greater_no(&self) -> bool { false }
}
