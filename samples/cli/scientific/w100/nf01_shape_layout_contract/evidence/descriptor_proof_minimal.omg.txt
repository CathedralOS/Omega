data Layout { origin: u64; row_stride: u64; column_stride: u64; }
machine checked(backing: &[i32], index: u64) -> i32 requires index < backing.len { backing[index] }
machine last(backing: &[i32], layout: &Layout) -> i32
requires backing.len == 6
requires layout.origin == 0 && layout.row_stride == 1 && layout.column_stride == 2 {
    let offset: u64 = layout.origin + layout.row_stride + 2 * layout.column_stride;
    checked(backing, offset)
}
machine main() -> i32 {
    let backing: [i32; 6] = [1, 4, 2, 5, 3, 6];
    let layout: Layout = Layout { origin: 0, row_stride: 1, column_stride: 2 };
    last(&backing, &layout)
}
