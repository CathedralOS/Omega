machine change(items: &mut [i32]) -> bool requires items.len == 1 {
    items[0] = 7;
    true
}
machine main() crashes Trap {
    let mut items: [i32; 1] = [1];
    let first: bool = change(&mut items);
    let inverted: bool = !change(&mut items);
    transition first && !inverted && items[0] == 7 {
        true -> done()
        false -> bad()
    }
    state done() {}
    state bad() { crash Trap; }
}
