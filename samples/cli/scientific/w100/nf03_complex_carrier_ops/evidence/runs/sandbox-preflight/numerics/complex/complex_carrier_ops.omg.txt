// One carrier, two explicit floating formats. Arithmetic never allocates.
// Destination loans must be disjoint from borrowed inputs; copy explicitly
// before an in-place update. See complex_carrier_ops.api.txt for rounding,
// overflow and magnitude policy. No matrix shape or backing storage is owned here.
module complex::complex_carrier_ops;

use omega::language::core::float_operations;

pub data Complex<T [copy]> [copy] {
    real: T;
    imaginary: T;
}

// Errors are local to the complex magnitude operation, not a shared
// numeric-policy implementation. NF02 integration awaits its public interface.
pub data ComplexMagnitude<T [copy]> [copy] {
    case Value(value: T);
    case NonFiniteInput;
    case Overflow;
}

pub machine ComplexF32::add(destination: &mut Complex<f32>, left: &Complex<f32>, right: &Complex<f32>) {
    destination.real = left.real + right.real;
    destination.imaginary = left.imaginary + right.imaginary;
}

pub machine ComplexF32::multiply(destination: &mut Complex<f32>, left: &Complex<f32>, right: &Complex<f32>) {
    let real_real: f32 = left.real * right.real;
    let imaginary_imaginary: f32 = left.imaginary * right.imaginary;
    let real_imaginary: f32 = left.real * right.imaginary;
    let imaginary_real: f32 = left.imaginary * right.real;
    destination.real = real_real - imaginary_imaginary;
    destination.imaginary = real_imaginary + imaginary_real;
}

pub machine ComplexF32::conjugate(destination: &mut Complex<f32>, input: &Complex<f32>) {
    destination.real = input.real;
    destination.imaginary = F32::negate(input.imaginary);
}

// Norm means squared Euclidean norm; magnitude has a separate name.
pub machine ComplexF32::norm_squared(input: &Complex<f32>) -> f32 {
    let real_squared: f32 = input.real * input.real;
    let imaginary_squared: f32 = input.imaginary * input.imaginary;
    transition { _ -> (real_squared + imaginary_squared) }
}

// Scaling before squaring avoids avoidable intermediate overflow/underflow.
// It cannot represent a true magnitude beyond the destination format.
pub machine ComplexF32::try_magnitude(input: &Complex<f32>) -> ComplexMagnitude<f32> {
    let real_finite: bool = F32::is_finite(input.real);
    let imaginary_finite: bool = F32::is_finite(input.imaginary);
    transition real_finite && imaginary_finite {
        true -> absolute(input.real, input.imaginary)
        _ -> (ComplexMagnitude<f32>::NonFiniteInput)
    }
    state absolute(real: f32, imaginary: f32) {
        let mut real_absolute: f32 = real;
        let mut imaginary_absolute: f32 = imaginary;
        transition real < 0.0 { true -> negative_real(real, imaginary) _ -> absolute_imaginary(real_absolute, imaginary_absolute) }
    }
    state negative_real(real: f32, imaginary: f32) {
        transition { _ -> absolute_imaginary(F32::negate(real), imaginary) }
    }
    state absolute_imaginary(real: f32, imaginary: f32) {
        transition imaginary < 0.0 { true -> scale(real, F32::negate(imaginary)) _ -> scale(real, imaginary) }
    }
    state scale(real: f32, imaginary: f32) {
        transition real >= imaginary { true -> ordered(real, imaginary) _ -> ordered(imaginary, real) }
    }
    state ordered(high: f32, low: f32) {
        transition high == 0.0 { true -> (ComplexMagnitude<f32>::Value { value: 0.0 }) _ -> ratio(high, low) }
    }
    state ratio(high: f32, low: f32) {
        let ratio: f32 = low / high;
        let square: f32 = ratio * ratio;
        let factor: f32 = F32::square_root(1.0 + square);
        let magnitude: f32 = high * factor;
        let finite: bool = F32::is_finite(magnitude);
        transition finite { true -> (ComplexMagnitude<f32>::Value { value: magnitude }) _ -> (ComplexMagnitude<f32>::Overflow) }
    }
}
