// Source-only companion: console removed; failed numerical assertion crashes Trap.
// NF12 independent fixed-shape scientific customer. The rank-two descriptor
// integration awaits NF01; these private three-row machines do not define it.
use omega::language::core::float_operations;
use triangular_solves;

machine near(actual: f64, expected: f64) -> bool {
    abs(actual - expected) <= 0.000000000001
}

data Main {
    matrix: [f64; 9];
    rhs: [f64; 6];
}
machine forward_three(matrix: &[f64; 9], rhs: &mut [f64; 6], offset: u64) -> bool
requires offset <= 3 {
    transition matrix[0] == 0.0 || matrix[4] == 0.0 || matrix[8] == 0.0 {
        true -> rejected()
        false -> row0(matrix, rhs, offset)
    }
    state rejected() -> bool { false }
    state row0(matrix: &[f64; 9], rhs: &mut [f64; 6], offset: u64) -> bool
    requires offset <= 3 {
        transition triangular_try_pivot(rhs[offset + 0], matrix[0], false) {
            TriangularPivotResult::Solved { value } -> store0(matrix, rhs, offset, value)
            TriangularPivotResult::ZeroPivot -> rejected_pivot()
        }
    }
    state store0(matrix: &[f64; 9], rhs: &mut [f64; 6], offset: u64, value: f64) -> bool
    requires offset <= 3 {
        rhs[offset + 0] = value;
        transition { _ -> row1(matrix, rhs, offset) }
    }
    state row1(matrix: &[f64; 9], rhs: &mut [f64; 6], offset: u64) -> bool
    requires offset <= 3 {
        transition triangular_try_pivot(rhs[offset + 1] - matrix[1] * rhs[offset + 0], matrix[4], false) {
            TriangularPivotResult::Solved { value } -> store1(matrix, rhs, offset, value)
            TriangularPivotResult::ZeroPivot -> rejected_pivot()
        }
    }
    state store1(matrix: &[f64; 9], rhs: &mut [f64; 6], offset: u64, value: f64) -> bool
    requires offset <= 3 {
        rhs[offset + 1] = value;
        transition { _ -> row2(matrix, rhs, offset) }
    }
    state row2(matrix: &[f64; 9], rhs: &mut [f64; 6], offset: u64) -> bool
    requires offset <= 3 {
        transition triangular_try_pivot(rhs[offset + 2] - matrix[2] * rhs[offset + 0] - matrix[5] * rhs[offset + 1], matrix[8], false) {
            TriangularPivotResult::Solved { value } -> store2(matrix, rhs, offset, value)
            TriangularPivotResult::ZeroPivot -> rejected_pivot()
        }
    }
    state store2(matrix: &[f64; 9], rhs: &mut [f64; 6], offset: u64, value: f64) -> bool
    requires offset <= 3 {
        rhs[offset + 2] = value;
        transition { _ -> (true) }
    }
    state rejected_pivot() -> bool { false }
}

machine Main::main(main: &mut Main) crashes Trap {
    main.matrix[0] = 2.0;
    main.matrix[1] = 3.0;
    main.matrix[2] = (0.0 - 1.0);
    main.matrix[3] = 0.0;
    main.matrix[4] = 1.0;
    main.matrix[5] = 4.0;
    main.matrix[6] = 0.0;
    main.matrix[7] = 0.0;
    main.matrix[8] = 2.0;
    main.rhs[0] = 2.0;
    main.rhs[1] = 1.0;
    main.rhs[2] = (0.0 - 3.0);
    main.rhs[3] = (0.0 - 2.0);
    main.rhs[4] = 0.0;
    main.rhs[5] = 17.0;
    let first: bool = forward_three(&main.matrix, &mut main.rhs, 0);
    let second: bool = forward_three(&main.matrix, &mut main.rhs, 3);
    transition first
        && second
        && near(main.rhs[0], 1.0)
        && near(main.rhs[1], (0.0 - 2.0))
        && near(main.rhs[2], 3.0)
        && near(main.rhs[3], (0.0 - 1.0))
        && near(main.rhs[4], 3.0)
        && near(main.rhs[5], 2.0)
        && near(main.matrix[0] * main.rhs[0], 2.0)
        && near(main.matrix[1] * main.rhs[0] + main.matrix[4] * main.rhs[1], 1.0)
        && near(main.matrix[2] * main.rhs[0] + main.matrix[5] * main.rhs[1] + main.matrix[8] * main.rhs[2], (0.0 - 3.0))
        && near(main.matrix[0] * main.rhs[3], (0.0 - 2.0))
        && near(main.matrix[1] * main.rhs[3] + main.matrix[4] * main.rhs[4], 0.0)
        && near(main.matrix[2] * main.rhs[3] + main.matrix[5] * main.rhs[4] + main.matrix[8] * main.rhs[5], 17.0) {
        true -> singular_control(main)
        false -> bad(main)
    }
    // A zero at the LAST pivot must be detected before the FIRST RHS write.
    state singular_control(main: &mut Main) crashes Trap {
        main.matrix[8] = 0.0;
        main.rhs[0] = 2.0;
        main.rhs[1] = 1.0;
        main.rhs[2] = 0.0 - 3.0;
        let solved: bool = forward_three(&main.matrix, &mut main.rhs, 0);
        transition !solved && main.rhs[0] == 2.0 && main.rhs[1] == 1.0 && main.rhs[2] == (0.0 - 3.0) {
            true -> good(main)
            false -> bad(main)
        }
    }
    state good(main: &mut Main) crashes Trap {
    }
    state bad(main: &mut Main) crashes Trap {
        crash Trap;
    }
}
