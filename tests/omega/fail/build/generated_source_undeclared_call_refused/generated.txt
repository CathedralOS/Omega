public data GeneratedPair {
    left: i32;
    right: i32;
}

public machine generated_offset() -> i32 {
    seed() + 1
}

public machine generated_check() {
    let pair: GeneratedPair = GeneratedPair { left: seed(), right: generated_offset() };
    transition pair.left == 7 {
        true -> later(pair)
        _ -> done()
    }
    state later(pair: GeneratedPair) {
        let observed: i32 = generated_missing();
        transition observed == pair.right {
            true -> done()
            _ -> done()
        }
    }
    state done() {}
}
